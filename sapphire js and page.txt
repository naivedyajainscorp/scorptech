// ─────────────────────────────────────────────────────────────────────────────
// 1. AOS (Animate On Scroll) Initialization
// ─────────────────────────────────────────────────────────────────────────────

function initAOS() {
  if (typeof AOS === 'undefined') {
    console.warn('⚠️ AOS library not loaded');
    return;
  }
  
  AOS.init({
    duration: 300,
    once: true,
    offset: 50
  });
  
  console.log('✅ AOS initialized');
}

// ─────────────────────────────────────────────────────────────────────────────
// 2. Sapphire Hero Grid Generator
// ─────────────────────────────────────────────────────────────────────────────

function initializeSapphireHeroGrid() {
  const sapphireGrid = document.getElementById('sapphireCorporateGrid');
  if (!sapphireGrid) {
    console.log('ℹ️ Sapphire hero grid container not found');
    return;
  }

  sapphireGrid.innerHTML = ''; // Clear existing content

  // Calculate grid size based on screen size
  let gridCols = 10;
  let gridRows = 10;
  
  if (window.innerWidth < 576) {
    gridCols = 5;
    gridRows = 6;
  } else if (window.innerWidth < 768) {
    gridCols = 6;
    gridRows = 8;
  } else if (window.innerWidth < 992) {
    gridCols = 8;
    gridRows = 9;
  }

  const totalCells = gridCols * gridRows;

  // Create grid cells with Sapphire-specific classes
  for (let i = 0; i < totalCells; i++) {
    const gridCell = document.createElement('div');
    gridCell.className = 'sapphire-grid-cell';

    // Add pulse element
    const gridPulse = document.createElement('div');
    gridPulse.className = 'sapphire-grid-pulse';

    // Add random delay for staggered animation
    const randomDelay = Math.random() * 2;
    gridCell.style.animationDelay = `${randomDelay}s`;
    gridPulse.style.animationDelay = `${randomDelay + 1}s`;

    gridCell.appendChild(gridPulse);
    sapphireGrid.appendChild(gridCell);
  }

  console.log(`✅ Sapphire hero grid initialized (${gridCols}x${gridRows} = ${totalCells} cells)`);
}

// ─────────────────────────────────────────────────────────────────────────────
// 3. Module Showcase — Tab + Panel System
// ─────────────────────────────────────────────────────────────────────────────

/* ══════════════════════════════════════════════════════
   MODULE SHOWCASE — Tab + Panel System
   Replaces all fullscreen core-module modal behaviour.
   ══════════════════════════════════════════════════════ */

function initModuleShowcase() {
  const tabs      = document.querySelectorAll('.s-showcase-tab');
  const panels    = document.querySelectorAll('.s-showcase-panel');
  const showcase  = document.getElementById('module-showcase');
  const navEl     = document.getElementById('showcaseNav');

  if (!tabs.length || !panels.length) return;

  function activateModule(moduleId) {
    tabs.forEach(t => {
      const isActive = t.dataset.showcaseTarget === moduleId;
      t.classList.toggle('active', isActive);
      t.setAttribute('aria-selected', String(isActive));
    });

    panels.forEach(p => {
      p.classList.toggle('active', p.id === `showcase-${moduleId}`);
    });

    // Scroll active tab into view within the horizontal scroller
    const activeTab = document.querySelector(`.s-showcase-tab[data-showcase-target="${moduleId}"]`);
    activeTab?.scrollIntoView({ block: 'nearest', inline: 'center', behavior: 'smooth' });
  }

  // Tab bar clicks
  tabs.forEach(tab => {
    tab.addEventListener('click', () => activateModule(tab.dataset.showcaseTarget));
  });

  // Card "Explore" buttons → scroll to showcase + activate module
  document.querySelectorAll('[data-showcase-target]').forEach(btn => {
    // Skip tabs themselves (they already have the click handler above)
    if (btn.classList.contains('s-showcase-tab')) return;

    btn.addEventListener('click', () => {
      activateModule(btn.dataset.showcaseTarget);
      const navOffset = navEl ? navEl.offsetHeight : 0;
      const top = showcase.getBoundingClientRect().top + window.scrollY - 64 - navOffset;
      window.scrollTo({ top, behavior: 'smooth' });
    });
  });

  console.log('✅ Module Showcase initialized');
}

// ─────────────────────────────────────────────────────────────────────────────
// INITIALIZATION
// ─────────────────────────────────────────────────────────────────────────────

document.addEventListener('DOMContentLoaded', () => {
  initAOS();
  initializeSapphireHeroGrid();
  initModuleShowcase();

  // Reinitialize grid on window resize (debounced)
  let resizeTimeout;
  window.addEventListener('resize', () => {
    clearTimeout(resizeTimeout);
    resizeTimeout = setTimeout(() => {
      initializeSapphireHeroGrid();
    }, 250);
  }, { passive: true });

  console.log('✅ Sapphire page fully initialized');
});

// Remove loading class when page fully loads
window.addEventListener('load', () => {
  document.body.classList.remove('loading');
}, { passive: true });

/* 2) Animate-on-scroll with IntersectionObserver */
function initAnimateOnScroll() {
  const elements = $$('.animate-on-scroll');
  if (!elements.length || !('IntersectionObserver' in window)) return;

  const io = new IntersectionObserver((entries) => {
    entries.forEach((entry) => {
      if (entry.isIntersecting) entry.target.classList.add('visible');
    });
  }, { threshold: 0.1 });

  elements.forEach((el) => io.observe(el));
}

/* 3) Sapphire hero grid (responsive) */
function initSapphireHeroGrid(sapphireGrid) {
  if (!sapphireGrid) return;

  const gen = () => {
    sapphireGrid.innerHTML = '';

    let gridCols = 10, gridRows = 10;
    if (window.innerWidth <= 576) { gridCols = 5; gridRows = 6; }
    else if (window.innerWidth <= 768) { gridCols = 6; gridRows = 8; }

    const totalCells = gridCols * gridRows;
    for (let i = 0; i < totalCells; i++) {
      const gridCell = document.createElement('div');
      gridCell.className = 'sapphire-grid-cell';
      const gridPulse = document.createElement('div');
      gridPulse.className = 'sapphire-grid-pulse';
      const randomDelay = Math.random() * 2;

      gridCell.style.animationDelay = `${randomDelay}s`;
      gridPulse.style.animationDelay = `${randomDelay + 1}s`;

      gridCell.appendChild(gridPulse);
      sapphireGrid.appendChild(gridCell);
    }
  };

  gen();
  // store generator for external reuse
  sapphireGrid._regen = gen;
}

function initGridResize(sapphireGrid) {
  if (!sapphireGrid) return;
  let resizeTimeout;
  const onResize = () => {
    clearTimeout(resizeTimeout);
    resizeTimeout = setTimeout(() => {
      if (typeof sapphireGrid._regen === 'function') sapphireGrid._regen();
    }, 150);
  };
  window.addEventListener('resize', onResize, { passive: true });
}

/* 4) Mobile parallax tilt for phone mockup (guarded, respects visibility) */
function initMobileParallax(phoneContainer) {
  if (!phoneContainer) return;
  const section = document.querySelector('.mobile-usp-section');
  if (!section) return;

  const reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  if (reduce) return;

  const MAX_TILT_Y = 35;   // left-right
  const MAX_TILT_X = 8.5; // up-down
  const EASE = 0.1;

  let isInView = false;
  let sx = 0, sy = 0; // deltas from center
  let tx = 0, ty = 0; // eased rotation
  let cx = 0, cy = 0; // center
  let animId = null;
  let armed = false;

  const resetNeutral = () => {
    phoneContainer.style.transform =
      'perspective(800px) rotateX(0deg) rotateY(0deg) translateX(0px) translateY(0px) scale(1)';
    const shadow = phoneContainer.querySelector('.phone-shadow');
    if (shadow) {
      shadow.style.transform = 'translateX(-50%) translateY(0px) scale(1)';
      shadow.style.opacity = '0.25';
    }
  };

  const updateCenter = () => {
    const r = phoneContainer.getBoundingClientRect();
    cx = r.left + r.width / 2;
    cy = r.top + r.height / 2;
  };

  // Start flat and cache geometry
  resetNeutral();
  updateCenter();

  const visObserver = new IntersectionObserver((entries) => {
    entries.forEach((entry) => {
      isInView = entry.isIntersecting;
      if (isInView) {
        updateCenter();
        resetNeutral();
        armed = false;
        if (!animId) animId = requestAnimationFrame(tick);
      }
    });
  }, { threshold: 0.1 });

  visObserver.observe(section);

  section.addEventListener('pointermove', (e) => {
    armed = true;
    sx = e.clientX - cx;
    sy = e.clientY - cy;
    if (!animId) animId = requestAnimationFrame(tick);
  });

  window.addEventListener('resize', updateCenter, { passive: true });

  function tick() {
    // Normalize by viewport size
    const nx = sx / (window.innerWidth / 2);
    const ny = sy / (window.innerHeight / 2);

    // Targets (clamped)
    const targetRY = Math.max(-MAX_TILT_Y, Math.min(MAX_TILT_Y, nx * MAX_TILT_Y));
    const targetRX = Math.max(-MAX_TILT_X, Math.min(MAX_TILT_X, -ny * MAX_TILT_X));

    // Ease toward target
    tx += (targetRX - tx) * EASE;
    ty += (targetRY - ty) * EASE;

    if (isInView && armed) {
      const translateX = ty / 6;
      const translateY = -tx / 4;
      phoneContainer.style.transform =
        `perspective(800px)
          rotateX(${tx.toFixed(2)}deg)
          rotateY(${ty.toFixed(2)}deg)
          translateX(${translateX.toFixed(1)}px)
          translateY(${translateY.toFixed(1)}px)
          scale(1)`;

      const shadow = phoneContainer.querySelector('.phone-shadow');
      if (shadow) {
        const shadowOffset = Math.abs(ty) + Math.abs(tx);
        shadow.style.transform =
          `translateX(-50%) translateY(${(shadowOffset * 0.4).toFixed(1)}px)
           scale(${(1 + shadowOffset * 0.008).toFixed(3)})`;
        shadow.style.opacity = (0.25 - shadowOffset * 0.008).toFixed(3);
      }
    }

    animId = requestAnimationFrame(tick);
  }
}

      if (t) t.focus();

}

/* 7) Card tilt (reduced-motion guarded) */
function initCardTiltEffects() {
  const reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  if (reduce) return;

  const cards = $$('.core-modules-card[data-tilt]');
  cards.forEach((card) => {
    let rafId = null;
    card.style.setProperty('--glx', '50%');
    card.style.setProperty('--gly', '0%');

    const onMove = (e) => {
      const r = card.getBoundingClientRect();
      const x = e.clientX - r.left, y = e.clientY - r.top;
      const rx = Math.max(-3, Math.min(3, (y - r.height / 2) / 50));
      const ry = Math.max(-3, Math.min(3, (r.width / 2 - x) / 50));
      const glx = (x / r.width) * 100;
      const gly = Math.max(0, (y / r.height) * 100);

      if (rafId) cancelAnimationFrame(rafId);
      rafId = requestAnimationFrame(() => {
        card.style.transform = `perspective(900px) rotateX(${rx}deg) rotateY(${ry}deg)`;
        card.style.setProperty('--glx', glx + '%');
        card.style.setProperty('--gly', gly + '%');
      });
    };

    const reset = () => {
      if (rafId) cancelAnimationFrame(rafId);
      card.style.transform = 'perspective(900px) rotateX(0) rotateY(0)';
      card.style.setProperty('--glx', '50%');
      card.style.setProperty('--gly', '0%');
    };

    card.addEventListener('mousemove', onMove);
    card.addEventListener('mouseleave', reset);
    card.addEventListener('blur', reset, true);
  });
}

/* 8) Analytics: reveal on view */
function initAnalyticsReveals() {
  const reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  const scope = $('#saphx-analytics');
  if (!scope) return;

  const els = $$('[data-reveal]', scope);
  if (!els.length) return;

  if (!('IntersectionObserver' in window)) {
    els.forEach((el) => el.classList.add('reveal-in'));
    return;
  }

  const io = new IntersectionObserver((entries) => {
    entries.forEach((e) => {
      if (e.isIntersecting) {
        const d = parseInt(e.target.getAttribute('data-reveal-delay') || '0', 10);
        setTimeout(() => e.target.classList.add('reveal-in'), reduce ? 0 : d);
        io.unobserve(e.target);
      }
    });
  }, { threshold: 0.15 });

  els.forEach((el) => io.observe(el));
}

/* 9) Analytics: gentle 3D tilt (scoped, reduced-motion guarded) */
function initAnalyticsTilt3D() {
  const section = $('#saphx-analytics');
  const visual = section?.querySelector('.tilt3d');
  if (!visual) return;

  const reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  if (reduce) return;

  let raf = null;
  let tx = 0, ty = 0, ex = 0, ey = 0;
  const MAXX = 6;
  const MAXY = 10;

  const calc = (el, e) => {
    const r = el.getBoundingClientRect();
    const x = (e.clientX - r.left) / r.width - 0.5;
    const y = (e.clientY - r.top) / r.height - 0.5;
    tx = y * -MAXX; // rotateX
    ty = x *  MAXY; // rotateY
  };

  const loop = () => {
    ex += (tx - ex) * 0.12;
    ey += (ty - ey) * 0.12;
    visual.style.transform = `perspective(900px) rotateX(${ex.toFixed(2)}deg) rotateY(${ey.toFixed(2)}deg)`;
    raf = requestAnimationFrame(loop);
  };

  const onEnter = (e) => {
    calc(visual, e);
    if (!raf) raf = requestAnimationFrame(loop);
  };
  const onMove = (e) => calc(visual, e);
  const onLeave = () => { tx = 0; ty = 0; if (!raf) raf = requestAnimationFrame(loop); };

  visual.addEventListener('pointerenter', onEnter);
  visual.addEventListener('pointermove', onMove);
  visual.addEventListener('pointerleave', onLeave);
}

/* 10) IntelligenceAnalytics (kept behavior, tidied) */
class IntelligenceAnalytics {
  constructor() {
    this.isUserClick = false;
    this.userClickTimer = null;
    this.init();
  }

  init() {
    this.setupEventListeners();
    this.setupScrollSpy();
    this.setupResponsiveHandlers();

    // Initial active states for each pair
    this.setActiveNavPoint('multi-location', 'intelligence-nav-1');
    this.setActiveNavPoint('asset-tracking', 'intelligence-nav-2');
    this.setActiveNavPoint('cost-analytics', 'intelligence-nav-3');
  }

  setupEventListeners() {
    // Navigation point clicks for desktop scrolling
    $$('.nav-point').forEach((point) => {
      point.addEventListener('click', (e) => {
        e.preventDefault();
        const target = e.currentTarget.getAttribute('data-target');
        const navContainer = e.currentTarget.closest('.nav-points');
        const navId = navContainer.id;

        if (window.innerWidth >= 992) {
          this.handleDesktopNavClick(target, navId, e.currentTarget);
        }
      });
    });

    // Intelligence card clicks for mobile modal
    $$('.intelligence-card[data-modal]').forEach((card) => {
      card.addEventListener('click', (e) => {
        if (window.innerWidth < 992 && typeof bootstrap !== 'undefined') {
          const modalId = e.currentTarget.getAttribute('data-modal');
          const modalEl = document.getElementById(modalId);
          if (modalEl) new bootstrap.Modal(modalEl).show();
        }
      });
    });

    // Content area scroll listeners for each pair
    $$('.content-scrollable').forEach((scrollable) => {
      scrollable.addEventListener('scroll', () => {
        if (!this.isUserClick) {
          const contentArea = scrollable.closest('.content-area');
          const contentId = contentArea.id;
          this.handleContentScroll(scrollable, contentId);
        }
      }, { passive: true });
    });

    // Window resize handler
    window.addEventListener('resize', () => this.handleResize(), { passive: true });
  }

  setupScrollSpy() {
    // Custom scroll spy for each pair independently
    if (window.innerWidth >= 992) {
      this.observeSections('operations-content', 'intelligence-nav-1');
      this.observeSections('asset-content', 'intelligence-nav-2');
      this.observeSections('financial-content', 'intelligence-nav-3');
    }
  }

  observeSections(contentId, navId) {
    const contentArea = document.getElementById(contentId);
    if (!contentArea || !('IntersectionObserver' in window)) return;

    const scrollable = contentArea.querySelector('.content-scrollable');
    const sections = scrollable.querySelectorAll('.content-section');

    const observerOptions = {
      root: scrollable,
      rootMargin: '-20% 0px -60% 0px',
      threshold: 0.1
    };

    const observer = new IntersectionObserver((entries) => {
      if (this.isUserClick) return;
      entries.forEach((entry) => {
        if (entry.isIntersecting) {
          const sectionId = entry.target.id;
          this.setActiveNavPoint(sectionId, navId);
        }
      });
    }, observerOptions);

    sections.forEach((section) => observer.observe(section));
  }

  handleDesktopNavClick(target, navId) {
    this.isUserClick = true;

    if (this.userClickTimer) clearTimeout(this.userClickTimer);

    this.setActiveNavPoint(target, navId, true);

    const navContainer = document.getElementById(navId);
    const row = navContainer.closest('.row');
    const contentArea = row.querySelector('.content-area');
    const contentScrollable = contentArea.querySelector('.content-scrollable');
    const targetSection = contentScrollable.querySelector('#' + target);

    if (targetSection && contentScrollable) {
      const offsetTop = targetSection.offsetTop - contentScrollable.offsetTop - 20;
      contentScrollable.scrollTo({ top: offsetTop, behavior: 'smooth' });
    }

    this.userClickTimer = setTimeout(() => { this.isUserClick = false; }, 1000);
  }

  handleContentScroll(scrollable, contentId) {
    if (this.isUserClick) return;

    const sections = scrollable.querySelectorAll('.content-section');
    const scrollTop = scrollable.scrollTop;

    let navId = '';
    switch (contentId) {
      case 'operations-content': navId = 'intelligence-nav-1'; break;
      case 'asset-content': navId = 'intelligence-nav-2'; break;
      case 'financial-content': navId = 'intelligence-nav-3'; break;
    }

    let activeSection = null;
    sections.forEach((section) => {
      const sectionTop = section.offsetTop - scrollable.offsetTop;
      const sectionHeight = section.offsetHeight;
      if (scrollTop >= sectionTop - 100 && scrollTop < sectionTop + sectionHeight - 100) {
        activeSection = section.id;
      }
    });

    if (activeSection && navId) this.setActiveNavPoint(activeSection, navId, false);
  }

  setActiveNavPoint(targetId, navId) {
    const navContainer = document.getElementById(navId);
    if (!navContainer) return;

    navContainer.querySelectorAll('.nav-point').forEach((point) => {
      point.classList.remove('active');
    });

    const targetNavPoint = navContainer.querySelector(`[data-target="${targetId}"]`);
    if (targetNavPoint) targetNavPoint.classList.add('active');
  }

  handleResize() {
    if (window.innerWidth >= 992) {
      this.setupScrollSpy();
    } else {
      this.isUserClick = false;
    }
  }

  setupResponsiveHandlers() {
    const mq = window.matchMedia('(max-width: 991.98px)');
    mq.addEventListener('change', (e) => {
      if (e.matches) {
        this.isUserClick = false;
      } else {
        this.setupScrollSpy();
      }
    });
  }
}
